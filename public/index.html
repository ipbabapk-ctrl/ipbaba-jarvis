} /* Normal text response */ else { rawText = String( event.data ); } }catch(error){ console.error( "Live raw message read error:", error ); setLiveStatus( "LIVE DEBUG ERROR", "Could not read Google's realtime response." ); return; } /* ALWAYS SHOW RAW RESPONSE IN CONSOLE */ console.log( "========== GEMINI LIVE RAW MESSAGE ==========" ); console.log( rawText ); console.log( "==============================================" ); /* Empty message */ if( !rawText ){ console.warn( "Gemini Live sent an empty message." ); return; } /* Parse JSON */ let data; try{ data = JSON.parse( rawText ); }catch(error){ console.error( "Live JSON parse error:", error ); console.error( "RAW GEMINI MESSAGE:", rawText ); /* IMPORTANT: Do NOT silently ignore unknown non-JSON messages. */ setMode( "error", "LIVE RESPONSE" ); setLiveStatus( "LIVE RESPONSE", rawText.slice( 0, 180 ) ); return; } /* SHOW COMPLETE PARSED OBJECT */ console.log( "Gemini Live parsed object:", data ); /* ========================================== GOOGLE ERROR ========================================== */ if( data.error ){ const message = data.error.message || data.error.status || "Google Live API error."; console.error( "Gemini Live error:", data.error ); state.liveReady = false; setMode( "error", "LIVE ERROR" ); setLiveStatus( "LIVE ERROR", message ); showLiveError( message ); try{ socket.close(); }catch(_){} return; } } } /* ========================================== SESSION RESUMPTION ========================================== */ const sessionUpdate = content.sessionResumptionUpdate || content.session_resumption_update; if( sessionUpdate ){ console.log( "Gemini Live session update:", sessionUpdate ); const handle = sessionUpdate.newHandle || sessionUpdate.new_handle || sessionUpdate.handle || sessionUpdate.token; if( handle ){ resumeHandle = handle; console.log( "Gemini Live resume handle received." ); } } /* ========================================== TURN COMPLETE ========================================== */ if( content.turnComplete || content.turn_complete ){ setMode( "listening", "LISTENING" );
